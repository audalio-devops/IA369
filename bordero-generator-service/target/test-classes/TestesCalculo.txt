## 8️⃣ Executar Testes via powershell
# Todos os testes
mvn test

# Apenas testes unitários
mvn test -Dtest=*Test

# Apenas testes de integração
mvn test -Dtest=*IntegrationTest

# Com cobertura
mvn test jacoco:report

# Teste específico
mvn test -Dtest=DiaUtilServiceTest

# Teste específico com método
mvn test -Dtest=DiaUtilServiceTest#d2EmSextaDeveIrParaTerca
